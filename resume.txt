  <div class="container" role="main">
    <h1>Resume Skill Extractor</h1>
    <p>Upload your resume (PDF or text) and get your key skills extracted instantly.</p>
    <input type="file" id="fileInput" accept=".pdf,.txt" aria-label="Upload resume file" />
    <button id="extractBtn" disabled>Extract Skills</button>
    <div id="loading" class="loading" aria-live="polite" style="display:none;">Extracting skills...</div>
    <div id="errorMsg" class="error" aria-live="assertive" style="display:none;"></div>
    <div id="skillsContainer" class="skills" aria-label="Extracted skills"></div>
  </div>
  <footer>
    &copy; 2024 Resume Skill Extractor
  </footer>

  <script>
    (() => {
      const fileInput = document.getElementById('fileInput');
      const extractBtn = document.getElementById('extractBtn');
      const loadingDiv = document.getElementById('loading');
      const errorMsg = document.getElementById('errorMsg');
      const skillsContainer = document.getElementById('skillsContainer');

      let selectedFile = null;

      fileInput.addEventListener('change', (e) => {
        errorMsg.style.display = 'none';
        skillsContainer.innerHTML = '';
        selectedFile = e.target.files[0];
        extractBtn.disabled = !selectedFile;
      });

      extractBtn.addEventListener('click', async () => {
        if (!selectedFile) return;
        loadingDiv.style.display = 'block';
        errorMsg.style.display = 'none';
        skillsContainer.innerHTML = '';
        extractBtn.disabled = true;
        fileInput.disabled = true;

        try {
          const formData = new FormData();
          formData.append('resume', selectedFile);

          // Change this URL to your backend API endpoint after deployment
          const response = await fetch('https://your-backend-url/api/extract-skills', {
            method: 'POST',
            body: formData,
          });

          if (!response.ok) {
            throw new Error(`Server error: ${response.statusText} (${response.status})`);
          }

          const data = await response.json();

          if (!data.skills || !Array.isArray(data.skills) || data.skills.length === 0) {
            errorMsg.textContent = 'No skills could be extracted from the resume.';
            errorMsg.style.display = 'block';
          } else {
            skillsContainer.innerHTML = '';
            data.skills.forEach(skill => {
              const span = document.createElement('span');
              span.className = 'skill-tag';
              span.textContent = skill;
              skillsContainer.appendChild(span);
            });
          }
        } catch (error) {
          errorMsg.textContent = `Error: ${error.message}`;
          errorMsg.style.display = 'block';
        } finally {
          loadingDiv.style.display = 'none';
          extractBtn.disabled = false;
          fileInput.disabled = false;
        }
      });
    })();
  </script>
